import subprocess,os

            ('位图','mvp/figures/audit/B_违规_位图冒充矢量.pdf'),

    fo=sh(f\"pdffonts '{p}'\"); L=[l for l in fo.split(chr(10))[2:] if l.strip()]
    raw=L[0] if L else ''
    nimg=len([l for l in sh(f\"pdfimages -list '{p}'\").split(chr(10))[2:] if l.strip()])
    txt=sh(f\"pdftotext '{p}' - 2>/dev/null | tr -d ' \\\\n'\")
    v=[]
    if nimg>0: v.append('位图')
    if 'Type 3' in raw: v.append('Type3')
    if len(txt)<10: v.append('无文本层')
    print(f, '→', '不合规:'+'/'.join(v) if v else '合规')
"
